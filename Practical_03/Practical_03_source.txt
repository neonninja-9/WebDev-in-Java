Experiment : Create a basic JSP page to generate dynamic content, and use JSP scriptlets, expressions and declarations to embed Java code in the JSP.
CODE : 
<%-- File: index.jsp --%>
<%@ page language="java" contentType="text/html; charset=UTF-8"
        import="java.util.Date" %>
<%!
    // Declaration: method available to the whole page
    int factorial(int n) {
        return (n <= 1) ? 1 : n * factorial(n - 1);
    }
%>
<html>
<body>
  <h2>JSP Scripting Elements</h2>
 
  <p>Current date and time: <%= new Date() %></p>
 
  <%
     // Scriptlet: plain Java code
     int num = 5;
     int fact = factorial(num);
  %>
  <p>Factorial of <%= num %> is <%= fact %></p>
 
  <h3>Multiplication table of 7</h3>
  <table border="1">
  <% for (int i = 1; i <= 10; i++) { %>
    <tr><td>7 x <%= i %></td><td><%= 7 * i %></td></tr>
  <% } %>
  </table>
</body>
</html>
